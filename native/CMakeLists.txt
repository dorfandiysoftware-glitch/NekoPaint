cmake_minimum_required(VERSION 3.10)
project(nekocore C)
set(CMAKE_C_STANDARD 99)
# No fused multiply-add: results must match the Java reference bit for bit.
if(MSVC)
  # Plain SSE2 (no /arch:AVX*), so it also runs on 2009-2011 CPUs such as Nehalem Xeons.
  add_compile_options(/O2 /fp:precise)
else()
  add_compile_options(-O3 -ffp-contract=off)
endif()
if(WIN32)
  # Windows: plain C library for P/Invoke from the C# app, no JNI.
  set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
  add_library(nekocore SHARED src/stroke.c)
else()
  add_library(nekocore SHARED src/stroke.c src/jni.c)
endif()
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "Linux" OR APPLE)
  target_compile_definitions(nekocore PRIVATE NP_THREADS)
  find_package(Threads)
  if(Threads_FOUND)
    target_link_libraries(nekocore Threads::Threads)
  endif()
endif()
target_include_directories(nekocore PRIVATE include)
find_library(M_LIB m)
if(M_LIB)
  target_link_libraries(nekocore ${M_LIB})
endif()
